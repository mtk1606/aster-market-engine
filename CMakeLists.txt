cmake_minimum_required(VERSION 3.20)
project(aster_market VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASTER_FUZZ "Build Clang/libFuzzer verification harnesses" OFF)
option(ASTER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(aster STATIC src/itch.cpp src/book.cpp src/mapped_file.cpp src/stream_file.cpp src/session.cpp)
target_include_directories(aster PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(aster PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(aster-replay src/main.cpp)
target_link_libraries(aster-replay PRIVATE aster)
if(ASTER_SANITIZE)
  foreach(target aster aster-replay)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(aster-tests tests/test_market.cpp)
  target_link_libraries(aster-tests PRIVATE aster)
  add_test(NAME aster-unit COMMAND aster-tests)
  add_test(NAME aster-fixture
    COMMAND ${CMAKE_COMMAND} -E env ASTER_BINARY=$<TARGET_FILE:aster-replay>
      python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py)
  add_test(NAME aster-independent
    COMMAND ${CMAKE_COMMAND} -E env ASTER_BINARY=$<TARGET_FILE:aster-replay>
      python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/crosscheck.py)
  add_test(NAME aster-session
    COMMAND ${CMAKE_COMMAND} -E env ASTER_BINARY=$<TARGET_FILE:aster-replay>
      python3 ${CMAKE_CURRENT_SOURCE_DIR}/tests/session.py)
endif()

if(ASTER_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Fuzz targets require Clang")
  endif()
  if(NOT ASTER_SANITIZE)
    message(FATAL_ERROR "Set -DASTER_SANITIZE=ON to instrument the replay library")
  endif()
  foreach(fuzzer aster-fuzz-frames aster-fuzz-book)
    if(fuzzer STREQUAL "aster-fuzz-frames")
      add_executable(${fuzzer} fuzz/fuzz_frames.cpp)
    else()
      add_executable(${fuzzer} fuzz/fuzz_book.cpp)
    endif()
    target_link_libraries(${fuzzer} PRIVATE aster)
    target_compile_options(${fuzzer} PRIVATE -fsanitize=fuzzer)
    target_link_options(${fuzzer} PRIVATE -fsanitize=fuzzer)
  endforeach()
endif()
